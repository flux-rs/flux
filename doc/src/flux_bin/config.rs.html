1//! Loading of `.flux/config.toml`, the configuration of a `cargo flux` invocation.
2//!
3//! Configuration files are discovered and merged following the same rules cargo uses for
4//! `.cargo/config.toml`:
5//!
6//! * Starting from the current directory, `.flux/config.toml` is read in every ancestor directory,
7//!   followed by `$FLUX_HOME/config.toml` (`~/.flux/config.toml` by default) if it wasn't already
8//!   found.
9//! * Files closer to the current directory take precedence. Tables are merged key by key, arrays
10//!   are concatenated (with items from higher precedence files placed later), and any other value
11//!   is replaced.
12//! * Relative paths are resolved relative to the parent of the directory containing the file,
13//!   e.g., a path in `/project/.flux/config.toml` is relative to `/project`.
14//!
15//! Settings can be overridden for specific packages in `[package.<spec>]` tables, where `<spec>`
16//! is a package name, `<name>@<version>`, or `*` for all packages that are not workspace members.
17//! As in cargo's profile overrides, a package's settings are taken from its own table, then from
18//! `[package."*"]`, and then from the top level.
19//!

32/// The merged content of all configuration files.
33#[derive(Debug, Default)]
34pub struct FluxConfig {
35    pub build: Build,
36    /// Top-level settings, kept untyped to merge them with package overrides. `None` if there are
37    /// no configuration files.
38    settings: Option<ConfigValue>,
39    /// `[package.<spec>]` tables, indexed by spec.
40    packages: BTreeMap<String, PackageOverride>,
41}
42
43/// A `[package.<spec>]` table.
44#[derive(Debug)]
45struct PackageOverride {
46    settings: ConfigValue,
47    fluxflags: Option<Vec<String>>,
48}

50/// The configuration of a single package.
51#[derive(Debug)]
52pub struct PackageConfig {

67/// A list of strings written either as an array or as a space-separated string.
68#[derive(Clone, Debug, Deserialize)]
69#[serde(untagged)]
70pub enum StringList {
71    String(String),
72    List(Vec<String>),
73}
74
75impl StringList {
76    fn into_vec(self) -> Vec<String> {
77        match self {
78            StringList::String(s) => split_space_separated(&s),
79            StringList::List(list) => list,
80        }
81    }
82}

184impl Settings {
185    /// Translates the settings into flags for the driver.
186    pub fn to_flags(&self, target_dir: &Utf8Path) -> Vec<String> {
187        let mut flags = vec![];
188        if let Some(true) = self.cache.enabled {
189            let dir = self
190                .cache
191                .dir
192                .as_ref()
193                .map(Located::resolve)
194                .unwrap_or_else(|| target_dir.join("flux-cache"));
195            flags.push(format!("-Fcache={dir}"));
196        }
197        if let Some(v) = self.fixpoint.timeout {
198            flags.push(format!("-Ffixpoint-timeout={v}"));
199        }
200        if let Some(v) = self.diagnostics.verbose {
201            flags.push(format!("-Fflux-verbose={v}"));
202        }
203        if let Some(v) = self.diagnostics.suggestions {
204            flags.push(format!("-Fno-suggestions={}", !v));
205        }
206        if let Some(v) = self.diagnostics.rerun_hint {
207            flags.push(format!("-Frerun-hint={v}"));
208        }
209        if let Some(v) = self.stats.summary {
210            flags.push(format!("-Fsummary={v}"));
211        }
212        if let Some(v) = self.stats.timings {
213            flags.push(format!("-Ftimings={v}"));
214        }
215        if let Some(v) = self.stats.annots {
216            flags.push(format!("-Fannots={v}"));
217        }
218        if let Some(v) = &self.debug.log_dir {
219            flags.push(format!("-Flog-dir={}", v.resolve()));
220        }
221        if let Some(v) = self.debug.dump_constraint {
222            flags.push(format!("-Fdump-constraint={v}"));
223        }
224        if let Some(v) = self.debug.dump_fhir {
225            flags.push(format!("-Fdump-fhir={v}"));
226        }
227        if let Some(v) = self.debug.dump_rty {
228            flags.push(format!("-Fdump-rty={v}"));
229        }
230        if let Some(v) = self.debug.dump_checker_trace {
231            flags.push(format!("-Fdump-checker-trace={}", v.as_str()));
232        }
233        if let Some(v) = self.debug.catch_bugs {
234            flags.push(format!("-Fcatch-bugs={v}"));
235        }
236        flags
237    }
238}

240impl FluxConfig {
241    /// Discovers and loads the configuration for an invocation in the current directory.
242    pub fn load() -> anyhow::Result<FluxConfig> {
243        let cwd = Utf8PathBuf::try_from(env::current_dir()?)?;
244        let flux_home = Utf8PathBuf::try_from(flux_home_dir())?;
245        Self::load_files(&discover(&cwd, &flux_home))
246    }
247
248    /// Loads and merges the given files, ordered from highest to lowest precedence.
249    fn load_files(paths: &[Utf8PathBuf]) -> anyhow::Result<FluxConfig> {
250        let values = paths
251            .iter()
252            .rev()
253            .map(|path| read_file(path))
254            .collect::<anyhow::Result<Vec<_>>>()?;
255        match ConfigValue::merge_all(values)? {
256            Some(merged) => Self::from_value(merged),
257            None => Ok(FluxConfig::default()),
258        }
259    }
260
261    fn from_value(mut value: ConfigValue) -> anyhow::Result<FluxConfig> {
262        let build: Build = match value.remove(BUILD_KEY) {
263            Some(build) => {
264                build
265                    .deserialize_into()
266                    .context("invalid `[build]` table")?
267            }
268            None => Build::default(),
269        };
270        let packages = match value.remove(PACKAGE_KEY) {
271            Some(packages) => {
272                let ConfigValue::Table(packages, _) = packages else {
273                    bail!("{}: `package` must be a table", packages.location());
274                };
275                packages
276                    .into_iter()
277                    .map(|(spec, value)| {
278                        let package = PackageOverride::from_value(value)
279                            .with_context(|| format!("invalid `[package.\"{spec}\"]` table"))?;
280                        Ok((spec, package))
281                    })
282                    .collect::<anyhow::Result<_>>()?
283            }
284            None => BTreeMap::new(),
285        };
286        value.clone().deserialize_into::<Settings>()?;
287        Ok(FluxConfig { build, settings: Some(value), packages })
288    }
289
290    /// Returns the configuration for the package `name` at `version`.
291    pub fn for_package(
292        &self,
293        name: &str,
294        version: &str,
295        is_workspace_member: bool,
296    ) -> anyhow::Result<PackageConfig> {
297        let matching: Vec<_> = self
298            .packages
299            .iter()
300            .filter(|(spec, _)| spec_matches(spec, name, version))
301            .collect();
302        if matching.len() >= 2 {
303            bail!(
304                "multiple `[package]` tables in `.flux/config.toml` match package `{name}`: `{}` and \
305                 `{}`",
306                matching[0].0,
307                matching[1].0
308            );
309        }
310        let named = matching.first().map(|(_, package)| *package);
311        let all_non_members =
312            if is_workspace_member { None } else { self.packages.get(ALL_NON_MEMBERS) };
313
314        // Overrides in increasing order of precedence. Settings are merged field by field.
315        let mut settings: Vec<ConfigValue> = self.settings.iter().cloned().collect();
316        let mut fluxflags = vec![];
317        for package in [all_non_members, named].into_iter().flatten() {
318            settings.push(package.settings.clone());
319            if let Some(flags) = &package.fluxflags {
320                fluxflags = flags.clone();
321            }
322        }
323        let settings = match ConfigValue::merge_all(settings)? {
324            Some(settings) => settings.deserialize_into()?,
325            None => Settings::default(),
326        };
327        Ok(PackageConfig { settings, fluxflags })
328    }
329
330    /// Returns the specs of `[package.<spec>]` tables that match none of `packages`, given as pairs
331    /// of name and version.
332    pub fn unmatched_specs(&self, packages: &[(String, String)]) -> Vec<&str> {
333        self.packages
334            .keys()
335            .filter(|spec| {
336                *spec != ALL_NON_MEMBERS
337                    && !packages
338                        .iter()
339                        .any(|(name, version)| spec_matches(spec, name, version))
340            })
341            .map(String::as_str)
342            .collect()
343    }
344
345    /// Returns the extra flags passed to every package. Following cargo's rules for rustflags,
346    /// they come from the first source that is set among [`FLUX_ENCODED_FLAGS`], [`FLUXFLAGS`] and
347    /// `build.fluxflags`. Sources are not combined.
348    pub fn global_fluxflags(&self) -> Vec<String> {
349        if let Ok(flags) = env::var(FLUX_ENCODED_FLAGS) {
350            if flags.is_empty() {
351                vec![]
352            } else {
353                flags.split('\x1f').map(str::to_string).collect()
354            }
355        } else if let Ok(flags) = env::var(FLUXFLAGS) {
356            split_space_separated(&flags)
357        } else {
358            self.build
359                .fluxflags
360                .clone()
361                .map(StringList::into_vec)
362                .unwrap_or_default()
363        }
364    }
365}

367impl PackageOverride {
368    fn from_value(mut settings: ConfigValue) -> anyhow::Result<PackageOverride> {
369        let fluxflags = match settings.remove(FLUXFLAGS_KEY) {
370            Some(value) => Some(value.deserialize_into::<StringList>()?.into_vec()),
371            None => None,
372        };
373        settings.clone().deserialize_into::<Settings>()?;
374        Ok(PackageOverride { settings, fluxflags })
375    }
376}

378/// Whether `spec` is `name` or `name@version`.
379fn spec_matches(spec: &str, name: &str, version: &str) -> bool {
380    match spec.split_once('@') {
381        Some((spec_name, spec_version)) => spec_name == name && spec_version == version,
382        None => spec == name,
383    }
384}
385
386fn split_space_separated(s: &str) -> Vec<String> {
387    s.split(' ')
388        .map(str::trim)
389        .filter(|s| !s.is_empty())
390        .map(str::to_string)
391        .collect()
392}

394/// Returns the configuration files that apply to `cwd`, ordered from highest to lowest
395/// precedence.
396fn discover(cwd: &Utf8Path, flux_home: &Utf8Path) -> Vec<Utf8PathBuf> {
397    let mut paths: Vec<Utf8PathBuf> = cwd
398        .ancestors()
399        .map(|dir| dir.join(".flux").join("config.toml"))
400        .filter(|path| path.is_file())
401        .collect();
402    let home_config = flux_home.join("config.toml");
403    if home_config.is_file() && !paths.iter().any(|path| same_file(path, &home_config)) {
404        paths.push(home_config);
405    }
406    paths
407}
408
409fn same_file(a: &Utf8Path, b: &Utf8Path) -> bool {
410    match (fs::canonicalize(a), fs::canonicalize(b)) {
411        (Ok(a), Ok(b)) => a == b,
412        _ => a == b,
413    }
414}

416/// Reads a configuration file and validates it on its own, so invalid values are reported even if
417/// another file overrides them.
418fn read_file(path: &Utf8Path) -> anyhow::Result<ConfigValue> {
419    let value = ConfigValue::parse_file(path)?;
420    FluxConfig::from_value(value.clone())?;
421    Ok(value)
