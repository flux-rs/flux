1#![cfg_attr(flux, flux::defs {
2    // A layout *fits* a block when the block is aligned to `layout.align()` and its size
3    // lands between the size it was allocated with and the size actually handed back.
4    // `NonNull` carries one size index, so we collapse that interval and track a block as
5    // exactly the `layout.size()` it was allocated with; keeping only the upper bound would
6    // admit deallocating with a smaller layout than was allocated, which is UB.
7    // See: https://doc.rust-lang.org/std/alloc/trait.Allocator.html#memory-fitting
8    //
9    // Weaker than the "currently allocated via this allocator" the methods below require:
10    // provenance, liveness and allocator identity are not checked, and the alignment checked
11    // is the address's, not the one the block was allocated with. Those clauses are assumed.
12    fn layout_fits(p: NonNull, l: Layout) -> bool {
13        p.base == p.addr && l.size == p.size && p.addr % l.align == 0
14    }
15})]
16
17// Unused to rustc — `Layout` appears only in `spec` attributes — but Flux resolves the
18// `Layout` sort in `layout_fits` through this import.
19#[allow(unused_imports)]
20use core::{alloc::Layout, ptr::NonNull};
21
22use flux_attrs::*;

24#[extern_spec(core::alloc)]
25trait Allocator {
26    /// Core impl: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/mod.rs#L133
27    /// On success the block meets the size and alignment guarantees of `layout`. The real
28    /// block may be larger than `layout.size()`; we track it as exactly `layout.size()`,
29    /// which under-approximates its extent — see `layout_fits`. Initialization is not
30    /// tracked either, so reading out of a freshly allocated block type-checks.
31    #[spec(fn(&Self, layout: Layout[@l])
32        -> Result<NonNull<[u8]>{p: layout_fits(p, l)}, AllocError>)]
33    fn allocate(&self, layout: Layout) -> Result<NonNull<[u8]>, AllocError>;
34
35    /// Core impl: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/mod.rs#L150
36    /// Same contract as `allocate`; that the block is zeroed is not modelled.
37    #[spec(fn(&Self, layout: Layout[@l])
38        -> Result<NonNull<[u8]>{p: layout_fits(p, l)}, AllocError>)]
39    fn allocate_zeroed(&self, layout: Layout) -> Result<NonNull<[u8]>, AllocError>;
40
41    /// Core impl: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/mod.rs#L166
42    /// - `ptr` must denote a block of memory currently allocated via this allocator
43    ///   (assumed, not checked), and
44    /// - `layout` must fit that block of memory.
45    ///
46    /// Deallocating does not invalidate `ptr`, so a later use of it is not rejected.
47    #[spec(fn(&Self, ptr: NonNull<u8>[@p], layout: Layout[@l]) requires layout_fits(p, l))]
48    unsafe fn deallocate(&self, ptr: NonNull<u8>, layout: Layout);
49
50    /// Core impl: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/mod.rs#L206
51    /// - `ptr` must denote a block of memory currently allocated via this allocator (assumed,
52    ///   not checked),
53    /// - `old_layout` must fit that block (`new_layout` need not), and
54    /// - `new_layout.size()` must be greater than or equal to `old_layout.size()`.
55    ///
56    /// Note that `new_layout.align()` need not be the same as `old_layout.align()`.
57    #[spec(fn(&Self, ptr: NonNull<u8>[@p], old_layout: Layout[@old], new_layout: Layout[@new])
58        -> Result<NonNull<[u8]>{q: layout_fits(q, new)}, AllocError>
59        requires layout_fits(p, old) && new.size >= old.size)]
60    unsafe fn grow(
61        &self,
62        ptr: NonNull<u8>,
63        old_layout: Layout,
64        new_layout: Layout,
65    ) -> Result<NonNull<[u8]>, AllocError>;
66
67    /// Core impl: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/mod.rs#L269
68    /// Same contract as `grow`; that the extra capacity is zeroed is not modelled.
69    #[spec(fn(&Self, ptr: NonNull<u8>[@p], old_layout: Layout[@old], new_layout: Layout[@new])
70        -> Result<NonNull<[u8]>{q: layout_fits(q, new)}, AllocError>
71        requires layout_fits(p, old) && new.size >= old.size)]
72    unsafe fn grow_zeroed(
73        &self,
74        ptr: NonNull<u8>,
75        old_layout: Layout,
76        new_layout: Layout,
77    ) -> Result<NonNull<[u8]>, AllocError>;
78
79    /// Core impl: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/mod.rs#L333
80    /// - `ptr` must denote a block of memory currently allocated via this allocator (assumed,
81    ///   not checked),
82    /// - `old_layout` must fit that block (`new_layout` need not), and
83    /// - `new_layout.size()` must be smaller than or equal to `old_layout.size()`.
84    ///
85    /// Note that `new_layout.align()` need not be the same as `old_layout.align()`.
86    #[spec(fn(&Self, ptr: NonNull<u8>[@p], old_layout: Layout[@old], new_layout: Layout[@new])
87        -> Result<NonNull<[u8]>{q: layout_fits(q, new)}, AllocError>
88        requires layout_fits(p, old) && new.size <= old.size)]
89    unsafe fn shrink(
90        &self,
91        ptr: NonNull<u8>,
92        old_layout: Layout,
93        new_layout: Layout,
94    ) -> Result<NonNull<[u8]>, AllocError>;
95}
