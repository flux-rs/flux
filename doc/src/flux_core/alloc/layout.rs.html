1#![cfg_attr(flux, flux::defs(
2    fn pow2(x: int) -> bool { pow2bv(bv_int_to_bv64(x)) }
3    fn pow2bv(x: bitvec<64>) -> bool {
4        bv_and(x, bv_sub(x, bv_int_to_bv64(1))) == bv_int_to_bv64(0)
5    }
6    // For a layout to be valid, align must be a power of two and size must not overflow
7    // isize when rounded up to the nearest multiple of align.
8    //
9    // The rounding formula `(size + align - 1) & !(align - 1)` is maximized when just past the
10    // alignment boundary, and thus bounded by `size + align - 1`.
11    // See: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/layout.rs#L80-L93
12    fn is_valid_layout(s: int, a: int) -> bool {
13        (s + a - 1 <= isize::MAX) && a > 0 && pow2(a)
14    }
15
16    // See: https://github.com/rust-lang/rust/blob/dab8d9d1066c4c95008163c7babf275106ce3f32/library/core/src/alloc/layout.rs#L559-L589
17    fn is_valid_array_layout(n: int, size: int, align: int) -> bool {
18        size == 0 || size * n + align - 1 <= isize::MAX
19    }
20))]
