1#![cfg_attr(flux, flux::defs {
2    // The memory range [addr, addr + num_bytes) is entirely contained within
3    // the pointer's allocation. In our model this is: addr >= base (not before
4    // the start) and num_bytes <= size (num_bytes remaining bytes fit before
5    // the end). The `size >= 0` guard rules out pointers already past the end.
6    // See: https://doc.rust-lang.org/std/ptr/index.html#safety
7    fn dereferenceable(p: ptr, num_bytes: int) -> bool {
8        p.addr >= p.base && num_bytes <= p.size && p.size >= 0
9    }
10
11    // Non-null and dereferenceable for `num_bytes` bytes. This is the strict
12    // validity predicate with no ZST exception: null is never valid, even for
13    // zero-sized accesses. Used as a building block for `valid` below.
14    fn valid_no_zst(p: ptr, num_bytes: int) -> bool {
15        p.addr != 0 && dereferenceable(p, num_bytes)
16    }
17
18    // Validity for raw-pointer operations (ptr::read, ptr::write, etc.).
19    // The Rust docs note that read and write allow null pointers when the
20    // total access size is zero (i.e., T is a ZST); for all other accesses
21    // the pointer must be non-null and dereferenceable.
22    // See: https://doc.rust-lang.org/std/ptr/index.html#safety
23    fn valid(p: ptr, num_bytes: int) -> bool {
24        num_bytes == 0 || valid_no_zst(p, num_bytes)
25    }
26
27    fn addr_aligned(addr: int, alignment: int) -> bool { addr % alignment == 0 }
28
29    // The pointer's address is a multiple of `alignment`. Most read/write
30    // functions require proper alignment (align_of::<T>()); the notable
31    // exceptions are read_unaligned and write_unaligned.
32    // See: https://doc.rust-lang.org/std/ptr/index.html#alignment
33    fn aligned_to(p: ptr, alignment: int) -> bool { addr_aligned(p.addr, alignment) }
34
35    // The byte ranges [p.addr, p.addr + num_bytes) and [q.addr, q.addr + num_bytes)
36    // do not overlap.
37    fn nonoverlapping(p: ptr, q: ptr, num_bytes: int) -> bool {
38        p.addr + num_bytes <= q.addr || q.addr + num_bytes <= p.addr
39    }
40
41    // The pointer itself lies within its allocation: addr >= base and size >= 0.
42    // One-past-the-end pointers (size == 0) satisfy this — they are valid
43    // starting points for pointer arithmetic but cannot be dereferenced.
44    // Required as a precondition for all pointer arithmetic methods.
45    fn in_bounds(p: ptr) -> bool { dereferenceable(p, 0) }
46})]
47/// These specs on `core::ptr` allow flux to enforce 2 safety properties:
48/// 1. Spatial safety: A pointer may only be read or written if pointer is derived from
49/// a valid allocation and the entire access would fall within the bounds of that allocation.
50/// 2. Per-method UB contract safety: Some methods in `core::ptr` have additional safety
51/// requirements beyond spatial safety, such as alignment or non-overlapping.
52/// These specs allow flux to enforce those requirements as well.
53///
54/// These specs do NOT prove temporal safety.
55/// To prove temporal safety, Flux would need to tie the lifetimes of allocations to the pointers derived from them,
56/// which is currently out of scope.
57mod non_null;

61macro_rules! ptr_specs {
62    ($mutable:tt $(, $($extra:tt)*)?) => {
63        #[extern_spec(core::ptr)]
64        impl<T> *$mutable T {
65            #[spec(fn (me: *$mutable[@p] T, count: usize)
66                -> *$mutable[p.base, p.addr + count * T::size_of(), p.size - count * T::size_of()] T
67                    requires in_bounds(p) && count * T::size_of() <= p.size
68                    ensures addr_aligned(p.addr, T::align_of()) => addr_aligned(p.addr + count * T::size_of(), T::align_of())
69            )]
70            unsafe fn add(self, count: usize) -> Self;
71
72            #[spec(fn (me: *$mutable[@p] T, count: usize)
73                -> *$mutable[p.base, p.addr - count * T::size_of(), p.size + count * T::size_of()] T
74                    requires in_bounds(p) && count * T::size_of() <= p.addr - p.base
75                    ensures addr_aligned(p.addr, T::align_of()) => addr_aligned(p.addr - count * T::size_of(), T::align_of())
76            )]
77            unsafe fn sub(self, count: usize) -> Self;
78
79            #[spec(fn (me: *$mutable[@p] T, count: usize)
80                -> *$mutable[p.base, p.addr + count, p.size - count] T
81                    requires in_bounds(p) && count <= p.size)]
82            unsafe fn byte_add(self, count: usize) -> Self;
83
84            #[spec(fn (me: *$mutable[@p] T, count: usize)
85                -> *$mutable[p.base, p.addr - count, p.size + count] T
86                    requires in_bounds(p) && count <= p.addr - p.base)]
87            unsafe fn byte_sub(self, count: usize) -> Self;
88
89            /// Core impl: https://github.com/rust-lang/rust/blob/7517636f510adf0a797e10cf655c21c0eb0723fb/library/core/src/ptr/const_ptr.rs#L22
90            #[no_panic]
91            #[spec(fn (me: *$mutable[@p] T) -> bool[p.addr == 0])]
92            fn is_null(self) -> bool;
93
94            /// Core impl: https://github.com/rust-lang/rust/blob/7517636f510adf0a797e10cf655c21c0eb0723fb/library/core/src/ptr/const_ptr.rs#L349
95            #[spec(fn (me: *$mutable[@p] T, count: isize)
96                -> *$mutable[p.base, p.addr + count * T::size_of(), p.size - count * T::size_of()] T
97                    requires in_bounds(p) && count * T::size_of() <= p.size && p.addr + count * T::size_of() >= p.base
98                    ensures addr_aligned(p.addr, T::align_of()) => addr_aligned(p.addr + count * T::size_of(), T::align_of())
99            )]
100            unsafe fn offset(self, count: isize) -> Self;
101
102            /// Core impl: https://github.com/rust-lang/rust/blob/7517636f510adf0a797e10cf655c21c0eb0723fb/library/core/src/ptr/const_ptr.rs#L471
103            #[spec(fn (me: *$mutable[@p] T, count: isize)
104                -> *$mutable[p.base, p.addr + count, p.size - count] T
105                    requires in_bounds(p) && count <= p.size && p.addr + count >= p.base)]
106            unsafe fn byte_offset(self, count: isize) -> Self;
107
108            // - Both `self` and `origin` must be derived from a pointer to the same allocation,
109            //   and the memory range between them must be in bounds of that allocation.
110            //   (`p.base == op.base` captures same-allocation provenance; `addr >= base` for
111            //   each captures the in-bounds requirement from below.)
112            // - The distance between the pointers in bytes must be an exact multiple of
113            //   `size_of::<T>()`, so that it corresponds to a whole number of elements.
114            // - `T` must not be a ZST; the function panics via `assert!(0 < size_of::<T>())`.
115            // See: https://github.com/rust-lang/rust/blob/7517636f510adf0a797e10cf655c21c0eb0723fb/library/core/src/ptr/const_ptr.rs#L614-L628
116            #[spec(fn (me: *$mutable[@p] T, origin: *const[@op] T) -> isize[(p.addr - op.addr) / T::size_of()]
117                requires p.base == op.base &&
118                         in_bounds(p) && in_bounds(op) &&
119                         (p.addr - op.addr) % T::size_of() == 0 &&
120                         T::size_of() > 0)]
121            unsafe fn offset_from(self, origin: *const T) -> isize
122            where T: Sized;
123
124            // - All safety conditions of `offset_from` apply.
125            // - Additionally, `self` must be greater than or equal to `origin`.
126            // See: https://github.com/rust-lang/rust/blob/7517636f510adf0a797e10cf655c21c0eb0723fb/library/core/src/ptr/const_ptr.rs#L770
127            #[spec(fn (me: *$mutable[@p] T, origin: *const[@op] T) -> usize[(p.addr - op.addr) / T::size_of()]
128                requires p.base == op.base &&
129                         in_bounds(p) && in_bounds(op) &&
130                         p.addr >= op.addr &&
131                         (p.addr - op.addr) % T::size_of() == 0 &&
132                         T::size_of() > 0)]
133            unsafe fn offset_from_unsigned(self, origin: *const T) -> usize
134            where T: Sized;
135
136            /// Core impl: https://github.com/rust-lang/rust/blob/c871d09d1cc32a649f4c5177bb819646260ed120/library/core/src/ptr/const_ptr.rs#L1166
137            #[spec(fn (me: *$mutable[@p] T) -> T
138                requires valid(p, T::size_of()) && aligned_to(p, T::align_of()))]
139            unsafe fn read(self) -> T
140            where T: Sized;
141
142            // Core impl: https://github.com/rust-lang/rust/blob/7517636f510adf0a797e10cf655c21c0eb0723fb/library/core/src/ptr/const_ptr.rs#L48
143            #[spec(fn(me: *$mutable[@p] T) -> *$mutable[p.base, p.addr, p.size] U)]
144            fn cast<U>(self) -> *$mutable U;
145
146            $($($extra)*)?
147        }
148    };
149}
