1#![cfg_attr(flux, flux::defs {
2    // The pointer lies within its allocation: addr >= base (not before the start)
3    // and size >= 0. One-past-the-end pointers (size == 0) satisfy this — they
4    // are valid starting points for arithmetic but cannot be dereferenced.
5    fn nn_in_bounds(base: int, addr: int, size: int) -> bool {
6        addr >= base && size >= 0
7    }
8
9    // The range [addr, addr + num_bytes) fits inside the allocation.
10    fn nn_dereferenceable(base: int, addr: int, size: int, num_bytes: int) -> bool {
11        addr >= base && num_bytes <= size && size >= 0
12    }
13
14    // Validity for read/write. Like ptr::valid, but drops the `addr != 0` guard —
15    // NonNull's invariant covers that. The num_bytes == 0 ZST shortcut is retained:
16    // a zero-byte access is valid even when the pointer is outside the allocation.
17    fn nn_valid(base: int, addr: int, size: int, num_bytes: int) -> bool {
18        num_bytes == 0 || nn_dereferenceable(base, addr, size, num_bytes)
19    }
20
21    // The address is a multiple of `alignment`.
22    fn nn_aligned_to(addr: int, alignment: int) -> bool {
23        addr % alignment == 0
24    }
25})]
