3#[extern_spec(core::slice)]
4#[refined_by(idx: int, len: int)]
5struct Iter<'a, T>;
6
7#[extern_spec(core::slice)]
8impl<'a, T> Iter<'a, T> {
9    #[no_panic]
10    #[spec(fn(&Self[@it]) -> &[T][it.len - it.idx])]
11    fn as_slice(&self) -> &'a [T];
12}
13
14#[extern_spec(core::slice)]
15#[assoc(
16    fn size(x: Iter) -> int { x.len - x.idx }
17    fn done(x: Iter) -> bool { x.idx >= x.len }
18    fn step(x: Iter, y: Iter) -> bool {
19        x.len == y.len &&
20        (if x.idx < x.len { y.idx == x.idx + 1 } else { y.idx == x.idx })
21    }
22)]
23impl<'a, T> Iterator for Iter<'a, T> {
24    #[no_panic]
25    #[spec(fn(self: &mut Iter<T>[@curr_s]) -> Option<_>[curr_s.idx < curr_s.len]
26           ensures self: Iter<T>{next_s:
27               curr_s.len == next_s.len &&
28               (curr_s.idx < curr_s.len => next_s.idx == curr_s.idx + 1) &&
29               (curr_s.idx >= curr_s.len => next_s.idx == curr_s.idx)
30           })]
31    fn next(&mut self) -> Option<&'a T>;
32
33    /// Core impl: https://github.com/rust-lang/rust/blob/c871d09d1cc32a649f4c5177bb819646260ed120/library/core/src/iter/traits/iterator.rs#L3049

35    fn position<P>(&mut self, predicate: P) -> Option<usize>
36    where
37        Self: Sized,

39}
