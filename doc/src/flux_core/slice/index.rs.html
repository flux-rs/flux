7/// Extern Specs for `ops::Index` which delegate to SliceIndex::index //////////////////////////
8////////////////////////////////////////////////////////////////////////////////////////////////
9
10#[extern_spec(core::slice)]
11impl<T, I: SliceIndex<[T]>> ops::Index<I> for [T] {
12    #![assoc(
13        fn in_bounds(len: int, idx: I) -> bool {
14            <I as SliceIndex<[T]>>::in_bounds(idx, len)
15        }
16        fn output_pred(len: int, idx: I, out: <I as SliceIndex<[T]>>::Output) -> bool {
17            <I as SliceIndex<[T]>>::output_pred(idx, len, out)
18        }
19    )]
20    /// Delegates to `SliceIndex::index`, documented as panicking iff out of
21    /// bounds, so `#[no_panic]` is sound under the `in_bounds` precondition.
22    /// Core impl: https://github.com/rust-lang/rust/blob/c6a955468b025dbe3d1de3e8f3e30496d1fb7f40/library/core/src/slice/index.rs#L15
23    #[no_panic]
24    #[sig(fn(&Self[@len], {I[@idx] | <Self as ops::Index<I>>::in_bounds(len, idx)}) -> &I::Output{out: <I as SliceIndex<[T]>>::output_pred(idx, len, out)})]
25    fn index(&self, index: I) -> &I::Output;
26}

42#[extern_spec(core::slice)]
43#[flux::assoc(fn in_bounds(idx: Self, v: T) -> bool)]
44#[flux::assoc(fn output_pred(idx: Self, v: T, out: Self::Output) -> bool { true })]
45trait SliceIndex<T> {}

51#[extern_spec(core::slice)]
52#[flux::assoc(fn in_bounds(r: Self, len: int) -> bool { r.start <= r.end && r.end <= len })]
53#[flux::assoc(fn output_pred(r: Self, len: int, out: int) -> bool { out == r.end - r.start })]
54impl<T> SliceIndex<[T]> for ops::Range<usize> {}

63#[flux::assoc(fn output_pred(r: Self, len: int, out: int) -> bool { out == len - r.start })]
64impl<T> SliceIndex<[T]> for ops::RangeFrom<usize> {}
65
66/// A full range is always in bounds and yields the whole slice. Without this impl, `v[..]` cannot
67/// be resolved once a spec mentions `<I as SliceIndex<[T]>>::in_bounds` — which matters for any
68/// `Index`/`IndexMut` impl generic over `I: SliceIndex<[T]>`, since `RangeFull` can instantiate it.
69#[extern_spec(core::slice)]
70#[flux::assoc(fn in_bounds(r: Self, len: int) -> bool { true })]
71#[flux::assoc(fn output_pred(r: Self, len: int, out: int) -> bool { out == len })]
72impl<T> SliceIndex<[T]> for ops::RangeFull {}
