1use core::{iter::Step, ops};
2
3use flux_attrs::*;
4// The below are "default" implementations of the associated refinements
5// for the `Step` trait, that we put in so that types for which no explicit
6// implementation is given can be analyzed without Flux complaining about missing
7// implementations. Note that the implementations are "uninterpreted" to make verification
8// sound. However, you may get "false positives" if you use these defaults.
9
10defs! {
11    fn default_step_step_forward<T>(start: T, count: int) -> T;
12    fn default_step_size<T>(lo: T, hi: T) -> int;
13}
14
15/// We define the following associated refinements for the `Step` trait, which are then
16/// used to specify the API for the `Iterator` implementation for `Range<A>`.
17///  - `step_forward` computes the new value after stepping forward from `start` by `count`,
18///  - `size` computes the number of steps needed to go from `lo` to `hi
19#[extern_spec(core::iter)]
20#[assoc(
21    fn step_forward(start: Self, count: int) -> Self {
22        default_step_step_forward(start, count)
23    }
24    fn size(lo: Self, hi: Self) -> int {
25        default_step_size(lo, hi)
26    }
27)]
28trait Step {}
29
30#[extern_spec(core::iter)]
31#[assoc(
32    fn step_forward(start: int, count: int) -> int { start + count }
33    fn size(lo: int, hi: int) -> int { hi - lo }
34)]
35impl Step for usize {}
36
37#[extern_spec(core::iter)]
38#[assoc(
39    fn step_forward(start: int, count: int) -> int { start + count }
40    fn size(lo: int, hi: int) -> int { hi - lo }
41)]
42impl Step for i32 {}
43
44#[extern_spec(core::ops)]
45#[assoc(
46    fn valid_item(self: Range<A>, item: A) -> bool { self.start <= item && item < self.end }
47    fn size(self: Range<A>) -> int { <A as Step>::size(self.start, self.end) }
48    fn done(self: Range<A>) -> bool { <A as Step>::size(self.start, self.end) <= 0 }
49)]
50impl<A: Step> Iterator for ops::Range<A> {
51    #[spec(
52        fn(self: &mut Range<A>[@old]) -> Option<A[old.start]>[old.start < old.end]
53        ensures self: Range<A>{r:
54            (old.start < old.end => r.start == <A as Step>::step_forward(old.start, 1)) &&
55            (!(old.start < old.end) => r.start == old.start) &&
56            r.end == old.end
57        }
58    )]
59    fn next(&mut self) -> Option<A>;
60}
