1//! Running Flux on a program: `cargo x run <file> -- -Fdump-checker-trace=debug` from the
2//! repository root, so Flux and its sysroot are rebuilt when needed, exactly as on the command
3//! line. Its output is streamed as it is printed; Flux writes the trace to a file, and rustc the
4//! MIR of each body (`-Zdump-mir=renumber`: the body borrow checking, and so Flux, works on) to a
5//! directory.
6
7use std::{
8    io::{BufRead, BufReader},
9    path::Path,
10    process::Stdio,
11    time::{Duration, Instant},
12};
13
14use tokio::{
15    process::Command,
16    sync::{Mutex, mpsc::UnboundedSender},
17};
18
19use crate::protocol::RunResult;
20
21/// The program's file name, as shown in the output instead of its temporary path.
22const SRC_NAME: &str = "input.rs";
23/// Generous, because a run may first rebuild Flux and its sysroot.
24const TIMEOUT: Duration = Duration::from_secs(900);
25/// `cargo x run` copies the sysroot, and concurrent runs clash doing that: one run at a time.
26static RUN_LOCK: Mutex<()> = Mutex::const_new(());
27
28/// Variables `cargo run` sets for blick itself. They would leak into the `cargo x run`
29/// started here and make it see the wrong package or toolchain.
30fn is_outer_cargo_var(k: &str) -> bool {
31    matches!(
32        k,
33        "CARGO"
34            | "CARGO_MANIFEST_DIR"
35            | "CARGO_MANIFEST_PATH"
36            | "CARGO_CRATE_NAME"
37            | "CARGO_BIN_NAME"
38            | "CARGO_PRIMARY_PACKAGE"
39            | "RUSTUP_TOOLCHAIN"
40            | "RUST_RECURSION_COUNT"
41    ) || k.starts_with("CARGO_PKG_")
42}

44/// Runs Flux on `source` from the repository at `repo`. Every output line is sent to `lines` as
45/// it is printed. Dropping the future stops the run.
46pub async fn run(
47    repo: &Path,
48    source: &str,
49    extra_flags: &[String],
50    lines: &UnboundedSender<String>,
51) -> anyhow::Result<RunResult> {
52    let _guard = match RUN_LOCK.try_lock() {
53        Ok(guard) => guard,
54        Err(_) => {
55            let _ = lines.send("[waiting for another run to finish]".to_string());
56            RUN_LOCK.lock().await
57        }
58    };
59    let tmp = tempfile::Builder::new().prefix("flux-trace-").tempdir()?;
60    let src = tmp.path().join(SRC_NAME);
61    std::fs::write(&src, source)?;
62    let src_str = src.to_string_lossy().into_owned();
63    let log_dir = tmp.path().join("log");
64    let mir_dir = tmp.path().join("mir");
65    let mut args: Vec<String> = vec![
66        "x".into(),
67        "run".into(),
68        src_str.clone(),
69        "--".into(),
70        "--color=always".into(),
71        "-Fdump-checker-trace=debug".into(),
72        format!("-Flog-dir={}", log_dir.display()),
73        "-Zdump-mir=renumber".into(),
74        format!("-Zdump-mir-dir={}", mir_dir.display()),
75        "-Zmir-include-spans=on".into(),
76    ];
77    args.extend(extra_flags.iter().cloned());
78    let shown = |s: &str| s.replace(&src_str, SRC_NAME);
79    let _ = lines.send(shown(&format!("$ cargo {}", args.join(" "))));
80
81    let start = Instant::now();
82    // stderr into stdout, so the lines arrive in the order they were printed
83    let (reader, writer) = std::io::pipe()?;
84    let mut child = {
85        let mut cmd = Command::new("cargo");
86        cmd.args(&args)
87            .current_dir(repo)
88            .stdin(Stdio::null())
89            .stdout(writer.try_clone()?)
90            .stderr(writer)
91            .env("CARGO_TERM_COLOR", "always")
92            .kill_on_drop(true);
93        for (k, _) in std::env::vars_os() {
94            if let Some(k) = k.to_str()
95                && is_outer_cargo_var(k)
96            {
97                cmd.env_remove(k);
98            }
99        }
100        // `cmd` is dropped here, closing our ends of the pipe's write side
101        cmd.spawn()?
102    };
103
104    let (tx, mut rx) = tokio::sync::mpsc::unbounded_channel::<String>();
105    tokio::task::spawn_blocking(move || {
106        for line in BufReader::new(reader).split(b'\n') {
107            let Ok(line) = line else { break };
108            if tx
109                .send(String::from_utf8_lossy(&line).into_owned())
110                .is_err()
111            {
112                break;
113            }
114        }
115    });
116
117    let finished = tokio::time::timeout(TIMEOUT, async {
118        while let Some(line) = rx.recv().await {
119            let _ = lines.send(shown(&line));
120        }
121        child.wait().await
122    })
123    .await;
124    let code = match finished {
125        Ok(status) => Some(status?.code().unwrap_or(-1)),
126        Err(_) => {
127            let _ = child.kill().await;
128            let _ = lines.send(format!("[timed out after {}s]", TIMEOUT.as_secs()));
129            None
130        }
131    };
132    let elapsed = (start.elapsed().as_secs_f64() * 100.0).round() / 100.0;
133    let trace = std::fs::read(log_dir.join("checker"))
134        .map(|b| String::from_utf8_lossy(&b).into_owned())
135        .unwrap_or_default();
136    let mir = read_mir_dumps(&mir_dir);
137    Ok(RunResult { ok: code == Some(0), exit: code, trace, mir, elapsed: Some(elapsed) })
138}

140/// The `.mir` files in `dir`, in file name order; none if rustc wrote none.
141fn read_mir_dumps(dir: &Path) -> Vec<String> {
142    let Ok(entries) = std::fs::read_dir(dir) else { return vec![] };
143    let mut paths: Vec<_> = entries
144        .filter_map(|e| Some(e.ok()?.path()))
145        .filter(|p| p.extension().is_some_and(|e| e == "mir"))
146        .collect();
147    paths.sort();
148    paths
149        .iter()
150        .filter_map(|p| std::fs::read(p).ok())
151        .map(|b| String::from_utf8_lossy(&b).into_owned())
152        .collect()
153}
