1//! The messages between the server and the page. Their TypeScript types are generated from these
2//! definitions (see [`write_bindings`]) into `client/src/bindings`.
3
4use std::{fs, path::Path};
5
6use serde::{Deserialize, Serialize};
7use ts_rs::{Config, TS};
8
9/// `GET /api/info`, shown in the page's status line.
10#[derive(Serialize, TS)]
11pub struct Info {
12    pub command: String,
13    pub repo: String,
14    /// The folder of programs the page can open (see `GET /api/files`), if blick was given one.
15    pub dir: Option<String>,
16}
17
18/// `GET /api/files`: the programs the page can open.
19#[derive(Serialize, TS)]
20pub struct FileList {
21    /// The folder they are in, absolute.
22    pub root: String,
23    /// `/`-separated paths relative to `root`, sorted.
24    pub files: Vec<String>,
25}
26
27/// The first (and only) message the page sends on `/api/run-ws`.
28#[derive(Deserialize, TS)]
29pub struct RunRequest {
30    pub source: String,
31    #[serde(default)]
32    pub flags: Vec<String>,
33}
34
35/// What the server sends on `/api/run-ws`: every output line while `cargo x run` works, then the
36/// result.
37#[derive(Serialize, TS)]
38#[serde(tag = "type", rename_all = "lowercase")]
39pub enum ServerMsg {
40    Line { text: String },
41    Done(RunResult),
42}
43
44#[derive(Serialize, TS)]
45pub struct RunResult {
46    /// Whether `cargo x run` exited successfully.
47    pub ok: bool,
48    /// The exit code; `None` if there is none (a timeout, or the run did not start).
49    pub exit: Option<i32>,
50    /// The checker trace (`log/checker`); empty if Flux did not write one.
51    pub trace: String,
52    /// The MIR of each body, as rustc dumps it (`-Zdump-mir=renumber`), with spans.
53    pub mir: Vec<String>,
54    /// Seconds the run took.
55    pub elapsed: Option<f64>,
56}
57
58impl RunResult {
59    /// A run that failed before `cargo x run` could finish; the reason goes in the output.
60    pub fn failed() -> Self {
61        RunResult { ok: false, exit: None, trace: String::new(), mir: vec![], elapsed: None }
62    }
63}
64
65/// Writes the TypeScript types of the messages into `dir`, leaving files whose contents did not
66/// change untouched (so the client is not rebuilt for nothing).
67pub fn write_bindings(dir: &Path) -> anyhow::Result<()> {
68    fs::create_dir_all(dir)?;
69    let cfg = Config::new();
70    write_binding::<Info>(dir, &cfg)?;
71    write_binding::<FileList>(dir, &cfg)?;
72    write_binding::<RunRequest>(dir, &cfg)?;
73    write_binding::<ServerMsg>(dir, &cfg)?;
74    write_binding::<RunResult>(dir, &cfg)?;
75    Ok(())
76}
77
78fn write_binding<T: TS + 'static>(dir: &Path, cfg: &Config) -> anyhow::Result<()> {
79    let path = dir.join(T::output_path().expect("derived types have an output path"));
80    let contents = T::export_to_string(cfg)?;
81    if fs::read_to_string(&path).ok().as_deref() != Some(contents.as_str()) {
82        fs::write(&path, contents)?;
83    }
84    Ok(())
85}
