8use crate::{

10    SysrootConfig,
11};

13/// Builds the sysroot for the host into `<out-dir>/sysroot`, with the solvers bundled, and packs it
14/// into `<out-dir>/flux-<host>.tar.gz`. The archive contains a top-level `sysroot` directory, so
15/// extracting it in the Flux home gives the default sysroot.
16pub(crate) fn dist(args: &Dist, rust_fixpoint: bool) -> anyhow::Result<()> {
17    let out_dir = args
18        .out_dir
19        .clone()
20        .unwrap_or_else(|| workspace_root().join("target").join("dist"));
21    let sysroot = out_dir.join("sysroot");
22    // Don't leave an archive from an earlier run behind if this one fails.
23    let archive = out_dir.join(format!("flux-{}.tar.gz", host_triple()?));
24    remove_path(&archive)?;
25    let config = SysrootConfig {
26        profile: Profile::Release,
27        rust_fixpoint,
28        dst: sysroot.clone(),
29        build_libs: BuildLibs { force: false, libs: FluxLib::ALL },
30        // A release must not ship without a solver.
31        bundle_solvers: BundleSolvers::Yes { required: true },
32    };
33    install_sysroot(&config)?;
34
35    write_archive(&sysroot, &archive)
36}
37
38/// The target triple of the host, as reported by the pinned `rustc`.
39fn host_triple() -> anyhow::Result<String> {
40    let output = Command::new("rustc")
41        .arg("-vV")
42        .current_dir(workspace_root())
43        .output()
44        .map_err(|err| anyhow!("failed to run `rustc -vV`: {err}"))?;
45    String::from_utf8(output.stdout)?
46        .lines()
47        .find_map(|line| line.strip_prefix("host: "))
48        .map(str::to_string)
49        .ok_or_else(|| anyhow!("`rustc -vV` didn't report the host"))
50}

52/// Packs `sysroot` into the `.tar.gz` file `archive`, under a top-level `sysroot` directory.
53fn write_archive(sysroot: &Path, archive: &Path) -> anyhow::Result<()> {
54    eprintln!("$ tar -czf {} sysroot", archive.display());
55    let file = fs::File::create(archive)
56        .map_err(|err| anyhow!("failed to create `{}`: {err}", archive.display()))?;
57    let mut builder =
58        tar::Builder::new(flate2::write::GzEncoder::new(file, flate2::Compression::default()));
59    builder.append_dir_all("sysroot", sysroot)?;
60    builder.into_inner()?.finish()?;
61    Ok(())
62}
