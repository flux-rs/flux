15/// Whether to bundle prebuilt solvers into the sysroot.
16#[derive(Clone, Copy)]
17pub(crate) enum BundleSolvers {
18    /// Don't bundle solvers. Flux looks them up in `PATH`.
19    No,
20    /// Bundle the solvers that have a prebuilt binary for the host and run on it.
21    Yes {
22        /// Fail if a solver can't be bundled. Otherwise, it's skipped with a warning and Flux looks
23        /// it up in `PATH`.
24        required: bool,
25    },
26}
27
28/// A solver downloaded into the sysroot by `cargo x install --bundle-solvers`.
29struct Solver {
30    bundled: BundledSolver,
31    /// URL of the release archive containing the solver for the host, if there is one.
32    archive_url: fn() -> Option<String>,
33}
34
35const FIXPOINT: Solver = Solver { bundled: install::FIXPOINT, archive_url: fixpoint_archive_url };
36
37const Z3: Solver = Solver { bundled: install::Z3, archive_url: z3_archive_url };
38
39fn fixpoint_archive_url() -> Option<String> {
40    let asset = match (env::consts::OS, env::consts::ARCH) {
41        ("linux", "x86_64") => "fixpoint-x86_64-linux-gnu.tar.gz",
42        ("macos", "aarch64") => "fixpoint-aarch64-apple-darwin.tar.gz",
43        ("windows", "x86_64") => "fixpoint-x86_64-pc-windows.zip",
44        _ => return None,
45    };
46    Some(format!(
47        "https://github.com/ucsd-progsys/liquid-fixpoint/releases/download/nightly/{asset}"
48    ))
49}

51/// The z3 version CI tests with.
52const Z3_VERSION: &str = "4.15.3";
53
54fn z3_archive_url() -> Option<String> {
55    let platform = match (env::consts::OS, env::consts::ARCH) {
56        // Only the platforms with a prebuilt fixpoint, see `fixpoint_archive_url`
57        ("linux", "x86_64") => "x64-glibc-2.39",
58        ("macos", "aarch64") => "arm64-osx-13.7.6",
59        ("windows", "x86_64") => "x64-win",
60        _ => return None,
61    };
62    Some(format!(
63        "https://github.com/Z3Prover/z3/releases/download/z3-{Z3_VERSION}/z3-{Z3_VERSION}-{platform}.zip"
64    ))
65}

67/// Downloads fixpoint and z3 into `sysroot`. Flux looks up the solvers that aren't bundled in
68/// `PATH`.
69///
70/// z3 is only bundled together with a fixpoint that supports `--solver-path`. Otherwise, Flux would
71/// pass the bundled z3 to a fixpoint that can't take it.
72pub(crate) fn bundle(sysroot: &Path, required: bool) -> anyhow::Result<()> {
73    let Some(fixpoint) = download_solver(&FIXPOINT, sysroot, required)? else {
74        return skip(required, "not bundling `z3` without `fixpoint`");
75    };
76    if !supports_solver_path(&fixpoint) {
77        return skip(
78            required,
79            "not bundling `z3` because the prebuilt `fixpoint` doesn't support `--solver-path`",
80        );
81    }
82    download_solver(&Z3, sysroot, required)?;
83    Ok(())
84}
85
86/// Gives up on bundling a solver: an error if `required`, and a warning otherwise.
87fn skip(required: bool, reason: &str) -> anyhow::Result<()> {
88    if required {
89        return Err(anyhow!("{reason}"));
90    }
91    eprintln!("warning: {reason}, Flux will look it up in PATH");
92    Ok(())
93}

95/// Downloads `solver` into `sysroot` and returns its path. Returns `None`, or an error if
96/// `required`, if there is no prebuilt binary for the host or if it doesn't run on it.
97fn download_solver(
98    solver: &Solver,
99    sysroot: &Path,
100    required: bool,
101) -> anyhow::Result<Option<PathBuf>> {
102    let name = solver.bundled.name;
103    let Some(url) = (solver.archive_url)() else {
104        let reason =
105            format!("there is no prebuilt `{name}` for {}-{}", env::consts::ARCH, env::consts::OS);
106        skip(required, &reason)?;
107        return Ok(None);
108    };
109    let exe = bin_path(sysroot, name);
110    let archive = bin_dir(sysroot).join(url.rsplit('/').next().unwrap_or(name));
111    download(&url, &archive)?;
112    extract_executable(&archive, &exe)?;
113    remove_path(&archive)?;
114
115    if !solver.bundled.runs(sysroot) {
116        remove_path(&exe)?;
117        skip(required, &format!("the prebuilt `{name}` doesn't run on this machine"))?;
118        return Ok(None);
119    }
120    Ok(Some(exe))
121}

123/// Whether `fixpoint` supports `--solver-path`, i.e., lists it in its help.
124fn supports_solver_path(fixpoint: &Path) -> bool {
125    Command::new(fixpoint)
126        .arg("--help")
127        .stderr(Stdio::null())
128        .output()
129        .is_ok_and(|output| String::from_utf8_lossy(&output.stdout).contains("--solver-path"))
130}

132/// Extracts the file named like `exe` from the `.zip` or `.tar.gz` `archive` into `exe`.
133fn extract_executable(archive: &Path, exe: &Path) -> anyhow::Result<()> {
134    eprintln!("$ extract {} from {}", exe.display(), archive.display());
135    let name = exe.file_name().unwrap_or_default();
136    let not_found = || anyhow!("`{}` does not contain `{}`", archive.display(), name.display());
137    let file = fs::File::open(archive)?;
138    let mut out = fs::File::create(exe)?;
139    if archive.extension().is_some_and(|ext| ext == "zip") {
140        let mut zip = zip::ZipArchive::new(file)?;
141        let index = (0..zip.len())
142            .find(|&i| {
143                zip.by_index(i).is_ok_and(|entry| {
144                    entry.is_file()
145                        && entry
146                            .name()
147                            .is_ok_and(|path| Path::new(path.as_ref()).file_name() == Some(name))
148                })
149            })
150            .ok_or_else(not_found)?;
151        io::copy(&mut zip.by_index(index)?, &mut out)?;
152    } else {
153        let mut tar = tar::Archive::new(flate2::read::GzDecoder::new(file));
154        let mut entry = tar
155            .entries()?
156            .filter_map(Result::ok)
157            .find(|entry| {
158                entry.header().entry_type().is_file()
159                    && entry
160                        .path()
161                        .is_ok_and(|path| path.file_name() == Some(name))
162            })
163            .ok_or_else(not_found)?;
164        io::copy(&mut entry, &mut out)?;
165    }
166    #[cfg(unix)]
167    {
168        use std::os::unix::fs::PermissionsExt;
169        fs::set_permissions(exe, fs::Permissions::from_mode(0o755))?;
170    }
171    Ok(())
172}
