1//! Prebuilt solvers bundled into the sysroot by `cargo x install --bundle-solvers`.
2
3use std::{env, fs, io, path::Path};
4
5use anyhow::anyhow;
6use flux_sysroot::{
7    bin_dir, bin_path,
8    install::{self, download, remove_broken_solvers, remove_path, BundledSolver},
9};

11/// Whether to bundle prebuilt solvers into the sysroot.
12#[derive(Clone, Copy)]
13pub(crate) enum BundleSolvers {
14    /// Don't bundle solvers. Flux looks them up in `PATH`.
15    No,
16    /// Bundle the solvers that have a prebuilt binary for the host and run on it.
17    Yes {
18        /// Fail if a solver can't be bundled. Otherwise, it's skipped with a warning and Flux looks
19        /// it up in `PATH`.
20        required: bool,
21    },
22}
23
24/// A solver downloaded into the sysroot by `cargo x install --bundle-solvers`.
25struct Solver {
26    bundled: BundledSolver,
27    /// URL of the release archive containing the solver for the host, if there is one.
28    archive_url: fn() -> Option<String>,
29}
30
31const FIXPOINT: Solver = Solver { bundled: install::FIXPOINT, archive_url: fixpoint_archive_url };
32
33const Z3: Solver = Solver { bundled: install::Z3, archive_url: z3_archive_url };
34
35fn fixpoint_archive_url() -> Option<String> {
36    let asset = match (env::consts::OS, env::consts::ARCH) {
37        ("linux", "x86_64") => "fixpoint-x86_64-linux-gnu.tar.gz",
38        ("macos", "aarch64") => "fixpoint-aarch64-apple-darwin.tar.gz",
39        ("windows", "x86_64") => "fixpoint-x86_64-pc-windows.zip",
40        _ => return None,
41    };
42    Some(format!(
43        "https://github.com/ucsd-progsys/liquid-fixpoint/releases/download/nightly/{asset}"
44    ))
45}
46
47/// The z3 version CI tests with.
48const Z3_VERSION: &str = "4.15.3";
49
50fn z3_archive_url() -> Option<String> {
51    let platform = match (env::consts::OS, env::consts::ARCH) {
52        // Only the platforms with a prebuilt fixpoint, see `fixpoint_archive_url`
53        ("linux", "x86_64") => "x64-glibc-2.39",
54        ("macos", "aarch64") => "arm64-osx-13.7.6",
55        ("windows", "x86_64") => "x64-win",
56        _ => return None,
57    };
58    Some(format!(
59        "https://github.com/Z3Prover/z3/releases/download/z3-{Z3_VERSION}/z3-{Z3_VERSION}-{platform}.zip"
60    ))
61}

63/// Downloads fixpoint and z3 into `sysroot`, and removes the ones that don't run on this machine,
64/// see [`remove_broken_solvers`]. Flux looks up the solvers that aren't bundled in `PATH`.
65pub(crate) fn bundle(sysroot: &Path, required: bool) -> anyhow::Result<()> {
66    download_solver(&FIXPOINT, sysroot, required)?;
67    download_solver(&Z3, sysroot, required)?;
68    remove_broken_solvers(sysroot, required)
69}
70
71/// Gives up on bundling a solver: an error if `required`, and a warning otherwise.
72fn skip(required: bool, reason: &str) -> anyhow::Result<()> {
73    if required {
74        return Err(anyhow!("{reason}"));
75    }
76    eprintln!("warning: {reason}, Flux will look it up in PATH");
77    Ok(())
78}
79
80/// Downloads `solver` into `sysroot`. Skips it, or errors if `required`, if there is no prebuilt
81/// binary for the host.
82fn download_solver(solver: &Solver, sysroot: &Path, required: bool) -> anyhow::Result<()> {
83    let name = solver.bundled.name;
84    let Some(url) = (solver.archive_url)() else {
85        let reason =
86            format!("there is no prebuilt `{name}` for {}-{}", env::consts::ARCH, env::consts::OS);
87        return skip(required, &reason);
88    };
89    let exe = bin_path(sysroot, name);
90    let archive = bin_dir(sysroot).join(url.rsplit('/').next().unwrap_or(name));
91    download(&url, &archive)?;
92    extract_executable(&archive, &exe)?;
93    remove_path(&archive)
94}

96/// Extracts the file named like `exe` from the `.zip` or `.tar.gz` `archive` into `exe`.
97fn extract_executable(archive: &Path, exe: &Path) -> anyhow::Result<()> {
98    eprintln!("$ extract {} from {}", exe.display(), archive.display());
99    let name = exe.file_name().unwrap_or_default();
100    let not_found = || anyhow!("`{}` does not contain `{}`", archive.display(), name.display());
101    let file = fs::File::open(archive)?;
102    let mut out = fs::File::create(exe)?;
103    if archive.extension().is_some_and(|ext| ext == "zip") {
104        let mut zip = zip::ZipArchive::new(file)?;
105        let index = (0..zip.len())
106            .find(|&i| {
107                zip.by_index(i).is_ok_and(|entry| {
108                    entry.is_file()
109                        && entry
110                            .name()
111                            .is_ok_and(|path| Path::new(path.as_ref()).file_name() == Some(name))
112                })
113            })
114            .ok_or_else(not_found)?;
115        io::copy(&mut zip.by_index(index)?, &mut out)?;
116    } else {
117        let mut tar = tar::Archive::new(flate2::read::GzDecoder::new(file));
118        let mut entry = tar
119            .entries()?
120            .filter_map(Result::ok)
121            .find(|entry| {
122                entry.header().entry_type().is_file()
123                    && entry
124                        .path()
125                        .is_ok_and(|path| path.file_name() == Some(name))
126            })
127            .ok_or_else(not_found)?;
128        io::copy(&mut entry, &mut out)?;
129    }
130    #[cfg(unix)]
131    {
132        use std::os::unix::fs::PermissionsExt;
133        fs::set_permissions(exe, fs::Permissions::from_mode(0o755))?;
134    }
135    Ok(())
136}
