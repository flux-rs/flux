1//! Shared sysroot utilities for `flux-bin`, `xtask`, and `tests`.

3use std::{
4    collections::BTreeMap,
5    env,
6    path::{Path, PathBuf},
7};

9use serde::{Deserialize, Serialize};
10
11/// Index of the sysroot, recording the name every artifact was given when it was copied in.
12///
13/// Written by `xtask`; read by `flux`, the test runner, and `flux-driver` to resolve the sysroot
14/// crates by path instead of by name.
15#[derive(Serialize, Deserialize, Default)]
16pub struct SysrootManifest {
17    #[serde(default)]
18    pub crates: BTreeMap<String, SysrootCrate>,
19}
20
21/// File names are relative to the sysroot directory.
22#[derive(Serialize, Deserialize, Default, Clone)]
23pub struct SysrootCrate {
24    #[serde(default, skip_serializing_if = "Option::is_none")]
25    pub rlib: Option<String>,
26    #[serde(default, skip_serializing_if = "Option::is_none")]
27    pub rmeta: Option<String>,
28    #[serde(default, skip_serializing_if = "Option::is_none")]
29    pub fluxmeta: Option<String>,
30    #[serde(default, skip_serializing_if = "Option::is_none")]
31    pub dylib: Option<String>,
32    /// Whether this crate's specs are injected as force externs under `-Fstd-extern-specs`.
33    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
34    pub extern_spec: bool,
35}
36
37impl SysrootManifest {
38    fn read(sysroot: &Path) -> Option<Self> {
39        let content = std::fs::read_to_string(sysroot.join(SYSROOT_MANIFEST)).ok()?;
40        toml::from_str(&content).ok()
41    }
42
43    /// Libraries get both their `.rmeta` and their `.rlib`: since cargo stopped embedding
44    /// metadata in rlibs, the former is the only copy of the metadata and the latter is still
45    /// needed to link.
46    pub fn extern_args(sysroot: &Path) -> Vec<String> {
47        let Some(manifest) = Self::read(sysroot) else { return vec![] };
48        let mut args = vec![];
49        for (name, krate) in &manifest.crates {
50            for file in [&krate.rmeta, &krate.rlib, &krate.dylib]
51                .into_iter()
52                .flatten()
53            {
54                args.push("--extern".to_string());
55                args.push(format!("{name}={}", sysroot.join(file).display()));
56            }
57        }
58        args
59    }
60
61    pub fn extern_specs(sysroot: &Path) -> Vec<(String, PathBuf)> {
62        let Some(manifest) = Self::read(sysroot) else { return vec![] };
63        manifest
64            .crates
65            .iter()
66            .filter(|(_, krate)| krate.extern_spec)
67            .filter_map(|(name, krate)| Some((name.clone(), sysroot.join(krate.rmeta.as_ref()?))))
68            .collect()
69    }
70}

72pub const SYSROOT_MANIFEST: &str = "sysroot.toml";

74/// Name of the environment variable used to override the Flux sysroot location.
75///
76/// The sysroot is a directory containing `flux-driver` and precompiled Flux libraries. When unset,
77/// the default is `~/.flux` (managed by `cargo x install`). Set this variable to redirect
78/// `flux` and `cargo-flux` to a different location, e.g., a custom or non-default install path.
79///
80/// During development, `cargo x build-sysroot` populates `<workspace-root>/sysroot/`. The test
81/// runner (`cargo x test`) automatically sets `FLUX_SYSROOT` to that directory so tests use the
82/// locally built artifacts instead of `~/.flux`.
83pub const FLUX_SYSROOT: &str = "FLUX_SYSROOT";
84
85/// Returns the path to the active Flux sysroot.
86///
87/// Both `cargo-flux` and `flux` call this at startup:
88/// - `cargo-flux` uses it to locate `flux-driver` and sets `RUSTC=<driver>` before invoking cargo.
89/// - `flux` uses it to locate `flux-driver` and passes `-L <sysroot>` so rustc can find the
90///   precompiled extern crates (`flux_rs`, `flux_attrs`).
91pub fn flux_sysroot_dir() -> PathBuf {
92    env::var_os(FLUX_SYSROOT).map_or_else(default_flux_sysroot_dir, PathBuf::from)
93}
94
95/// Returns `~/.flux`, the default Flux sysroot when [`FLUX_SYSROOT`] is not set.
96pub fn default_flux_sysroot_dir() -> PathBuf {
97    home::home_dir()
98        .expect("Couldn't find home directory")
99        .join(".flux")
100}
