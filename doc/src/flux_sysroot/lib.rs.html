13/// Index of the sysroot, recording the name every artifact was given when it was copied in.
14///
15/// Written by `xtask`; read by `flux`, the test runner, and `flux-driver` to resolve the sysroot
16/// crates by path instead of by name.
17#[derive(Serialize, Deserialize)]
18pub struct SysrootManifest {
19    /// Rust toolchain `flux-driver` was built with, and therefore must run with, e.g.,
20    /// `nightly-2026-11-21-x86_64-unknown-linux-gnu`.
21    pub toolchain: String,
22    /// Version of Flux the sysroot was built from.
23    pub version: FluxVersion,
24    #[serde(default)]
25    pub crates: BTreeMap<String, SysrootCrate>,
26}

31/// Error returned when the sysroot's manifest is missing or malformed. The error message suggests
32/// reinstalling Flux.
33#[derive(Debug)]
34pub struct InvalidSysroot {
35    sysroot: PathBuf,
36    reason: String,
37}
38
39impl fmt::Display for InvalidSysroot {
40    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
41        writeln!(f, "invalid Flux sysroot at `{}`", self.sysroot.display())?;
42        writeln!(f, "note: {}", self.reason)?;
43        write!(f, "help: try reinstalling Flux (see {INSTALL_GUIDE_URL})")
44    }
45}
46
47impl std::error::Error for InvalidSysroot {}

76/// File names are relative to the sysroot's `lib` directory.
77#[derive(Serialize, Deserialize, Default, Clone)]
78pub struct SysrootCrate {
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub rlib: Option<String>,
81    #[serde(default, skip_serializing_if = "Option::is_none")]
82    pub rmeta: Option<String>,
83    #[serde(default, skip_serializing_if = "Option::is_none")]
84    pub fluxmeta: Option<String>,
85    #[serde(default, skip_serializing_if = "Option::is_none")]
86    pub dylib: Option<String>,
87    /// Whether this crate's specs are injected as force externs under `-Fstd-extern-specs`.
88    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
89    pub extern_spec: bool,
90}

92impl SysrootManifest {
93    /// Reads the manifest of `sysroot`.
94    pub fn read(sysroot: &Path) -> Result<Self, InvalidSysroot> {
95        let path = sysroot.join(SYSROOT_MANIFEST);
96        let invalid = |reason| InvalidSysroot { sysroot: sysroot.to_path_buf(), reason };
97        let content = fs::read_to_string(&path)
98            .map_err(|err| invalid(format!("failed to read `{}`: {err}", path.display())))?;
99        toml::from_str(&content).map_err(|err| {
100            invalid(format!("failed to parse `{}`: {}", path.display(), err.message()))
101        })
102    }
103
104    /// Libraries get both their `.rmeta` and their `.rlib`: since cargo stopped embedding
105    /// metadata in rlibs, the former is the only copy of the metadata and the latter is still
106    /// needed to link.
107    pub fn extern_args(sysroot: &Path) -> Vec<String> {
108        let Ok(manifest) = Self::read(sysroot) else { return vec![] };
109        let mut args = vec![];
110        for (name, krate) in &manifest.crates {
111            for file in [&krate.rmeta, &krate.rlib, &krate.dylib]
112                .into_iter()
113                .flatten()
114            {
115                args.push("--extern".to_string());
116                args.push(format!("{name}={}", lib_dir(sysroot).join(file).display()));
117            }
118        }
119        args
120    }
121
122    pub fn extern_specs(sysroot: &Path) -> Vec<(String, PathBuf)> {
123        let Ok(manifest) = Self::read(sysroot) else { return vec![] };
124        manifest
125            .crates
126            .iter()
127            .filter(|(_, krate)| krate.extern_spec)
128            .filter_map(|(name, krate)| {
129                Some((name.clone(), lib_dir(sysroot).join(krate.rmeta.as_ref()?)))
130            })
131            .collect()
132    }
133}

135/// The sysroot is a self-contained directory with the following layout:
136///
137/// ```text
138/// sysroot.toml                     the manifest, see [`SysrootManifest`]
139/// bin/{cargo-flux,flux,flux-driver}
140/// lib/                             precompiled Flux libraries
141/// ```
142pub const SYSROOT_MANIFEST: &str = "sysroot.toml";
143
144/// Returns the directory of the sysroot holding the binaries.
145pub fn bin_dir(sysroot: &Path) -> PathBuf {
146    sysroot.join("bin")
147}
148
149/// Returns the directory of the sysroot holding the precompiled libraries.
150pub fn lib_dir(sysroot: &Path) -> PathBuf {
151    sysroot.join("lib")
152}
153
154/// Returns the path of the binary `name` in the sysroot.
155pub fn bin_path(sysroot: &Path, name: &str) -> PathBuf {
156    bin_dir(sysroot).join(format!("{name}{}", env::consts::EXE_SUFFIX))
157}

159/// Name of the environment variable used to override the Flux sysroot location.
160///
161/// When unset, the default is `$FLUX_HOME/sysroot` (managed by `cargo x install`). Set this variable
162/// to redirect `flux` and `cargo-flux` to a different sysroot.
163///
164/// During development, `cargo x build-sysroot` populates `<workspace-root>/sysroot/`. The test
165/// runner (`cargo x test`) automatically sets `FLUX_SYSROOT` to that directory so tests use the
166/// locally built artifacts instead of the installed ones.
167pub const FLUX_SYSROOT: &str = "FLUX_SYSROOT";
168
169/// Returns the path to the active Flux sysroot, i.e., [`FLUX_SYSROOT`] or `$FLUX_HOME/sysroot` if
170/// not set.
171pub fn flux_sysroot_dir() -> PathBuf {
172    env::var_os(FLUX_SYSROOT).map_or_else(default_flux_sysroot_dir, PathBuf::from)
173}
174
175/// Environment variable overriding the Flux home directory.
176///
177/// Like `CARGO_HOME` for cargo, the Flux home holds the user-level configuration file
178/// (`config.toml`) and the default installation of the sysroot (`sysroot/`). When unset, the
179/// default is `~/.flux`.
180pub const FLUX_HOME: &str = "FLUX_HOME";
181
182/// Returns the Flux home directory, i.e., [`FLUX_HOME`] or `~/.flux` if not set.
183pub fn flux_home_dir() -> PathBuf {
184    env::var_os(FLUX_HOME).map_or_else(
185        || {
186            home::home_dir()
187                .expect("Couldn't find home directory")
188                .join(".flux")
189        },
190        PathBuf::from,
191    )
192}
193
194/// Returns `$FLUX_HOME/sysroot`, the default Flux sysroot when [`FLUX_SYSROOT`] is not set.
195pub fn default_flux_sysroot_dir() -> PathBuf {
196    flux_home_dir().join("sysroot")
