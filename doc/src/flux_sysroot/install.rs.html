15/// Binaries in the sysroot that are linked from cargo's `bin` directory, which is where cargo looks
16/// for external subcommands and is usually on the `PATH`.
17pub const LINKED_BINARIES: &[&str] = &["cargo-flux", "flux"];

19/// A solver that can be bundled in the sysroot's `bin` directory.
20pub struct BundledSolver {
21    pub name: &'static str,
22    /// Argument that makes the solver print its version, used to check that it runs.
23    pub version_arg: &'static str,
24}
25
26impl BundledSolver {
27    /// Whether the solver bundled in `sysroot` runs on this machine.
28    pub fn runs(&self, sysroot: &Path) -> bool {
29        Command::new(bin_path(sysroot, self.name))
30            .arg(self.version_arg)
31            .stdout(Stdio::null())
32            .stderr(Stdio::null())
33            .status()
34            .is_ok_and(|status| status.success())
35    }
36}

45/// Returns the paths in cargo's `bin` directory the [`LINKED_BINARIES`] are linked from.
46pub fn binary_links() -> anyhow::Result<Vec<PathBuf>> {
47    let cargo_bin = home::cargo_home()?.join("bin");
48    Ok(LINKED_BINARIES
49        .iter()
50        .map(|name| cargo_bin.join(format!("{name}{}", env::consts::EXE_SUFFIX)))
51        .collect())
52}

54/// Links the [`LINKED_BINARIES`] in `sysroot` from cargo's `bin` directory, replacing whatever is
55/// there. The links point to `sysroot` itself rather than to the directory it resolves to, so they
56/// keep working when `sysroot` is replaced. Windows doesn't have unprivileged symlinks, so there
57/// the binaries are copied instead.
58pub fn link_binaries(sysroot: &Path) -> anyhow::Result<()> {
59    for (name, link) in LINKED_BINARIES.iter().zip(binary_links()?) {
60        if let Some(parent) = link.parent() {
61            create_dir(parent)?;
62        }
63        // Create the link next to its final location and rename it there, so it's replaced
64        // atomically.
65        let tmp = sibling_path(&link, "tmp");
66        remove_path(&tmp)?;
67        symlink_or_copy(&bin_path(sysroot, name), &tmp)?;
68        if cfg!(windows) {
69            // Windows doesn't replace an executable that is running, e.g., the `cargo-flux` doing
70            // the installation.
71            remove_executable(&link)?;
72        }
73        rename(&tmp, &link)?;
74    }
75    Ok(())
76}

78#[cfg(unix)]
79fn symlink_or_copy(target: &Path, link: &Path) -> anyhow::Result<()> {
80    eprintln!("$ ln -s {} {}", target.display(), link.display());
81    std::os::unix::fs::symlink(target, link).map_err(|err| {
82        anyhow!("failed to link `{}` to `{}`: {err}", link.display(), target.display())
83    })
84}
85
86#[cfg(not(unix))]
87fn symlink_or_copy(target: &Path, link: &Path) -> anyhow::Result<()> {
88    copy_file(target, link)
89}

91/// Whether `link` was created by [`link_binaries`] for `sysroot`. On Windows, where the binaries
92/// are copied, any file is assumed to be.
93fn is_link_to(link: &Path, sysroot: &Path) -> bool {
94    if cfg!(unix) {
95        fs::read_link(link).is_ok_and(|target| target.starts_with(bin_dir(sysroot)))
96    } else {
97        link.is_file()
98    }
99}

101/// Removes `sysroot` and the links [`link_binaries`] created for it.
102pub fn uninstall(sysroot: &Path) -> anyhow::Result<()> {
103    for link in binary_links()? {
104        if is_link_to(&link, sysroot) {
105            remove_executable(&link)?;
106        }
107    }
108    eprintln!("$ rm -rf {}", sysroot.display());
109    remove_path(sysroot)
110}

112/// Removes the executable `path`, which may be running. Windows doesn't remove a running
113/// executable, but does rename it, so there it's renamed to `<path>.old` if it can't be removed.
114/// That file is removed the next time.
115fn remove_executable(path: &Path) -> anyhow::Result<()> {
116    if !cfg!(windows) {
117        eprintln!("$ rm {}", path.display());
118        return remove_path(path);
119    }
120    let old = sibling_path(path, "old");
121    let _ = remove_path(&old);
122    if path.symlink_metadata().is_err() || remove_path(path).is_ok() {
123        return Ok(());
124    }
125    rename(path, &old)
126}

128/// Replaces `dst` with `src`. Where the platform and filesystem support it, the two are exchanged
129/// atomically. Elsewhere (e.g., on Windows), `dst` is moved out of the way first, so it briefly
130/// doesn't exist.
131pub fn replace_dir(src: &Path, dst: &Path) -> anyhow::Result<()> {
132    if !dst.exists() {
133        return rename(src, dst);
134    }
135    #[cfg(any(target_os = "linux", target_os = "macos"))]
136    {
137        use rustix::fs::{CWD, RenameFlags, renameat_with};
138        eprintln!("$ exchange {} {}", src.display(), dst.display());
139        if renameat_with(CWD, src, CWD, dst, RenameFlags::EXCHANGE).is_ok() {
140            // `src` now holds the old contents of `dst`
141            return remove_path(src);
142        }
143    }
144    let old = sibling_path(dst, "old");
145    remove_path(&old)?;
146    rename(dst, &old)?;
147    rename(src, dst)?;
148    remove_path(&old)
149}

151/// Downloads `url` into the file `dst` with `curl`.
152pub fn download(url: &str, dst: &Path) -> anyhow::Result<()> {
153    eprintln!("$ curl -o {} {url}", dst.display());
154    let status = Command::new("curl")
155        .args(["--proto", "=https", "--tlsv1.2", "-fsSL", "--retry", "3", "-o"])
156        .arg(dst)
157        .arg(url)
158        .status()
159        .map_err(|err| {
160            anyhow!("failed to run `curl`, which is needed to download `{url}`: {err}")
161        })?;
162    if !status.success() {
163        return Err(anyhow!("failed to download `{url}`: `curl` exited with {status}"));
164    }
165    Ok(())
166}

168pub fn rename(src: &Path, dst: &Path) -> anyhow::Result<()> {
169    eprintln!("$ mv {} {}", src.display(), dst.display());
170    fs::rename(src, dst).map_err(|err| {
171        anyhow!("failed to rename `{}` to `{}`: {err}", src.display(), dst.display())
172    })
173}

175/// Returns `path` with `.{suffix}` appended, e.g., `sysroot` becomes `sysroot.new`.
176pub fn sibling_path(path: &Path, suffix: &str) -> PathBuf {
177    let mut name = path.file_name().unwrap_or_default().to_os_string();
178    name.push(format!(".{suffix}"));
179    path.with_file_name(name)
180}

182/// Copies the file `src` to `dst`, or into `dst` if it's a directory.
183pub fn copy_file<S: AsRef<Path>, D: AsRef<Path>>(src: S, dst: D) -> anyhow::Result<()> {
184    let src = src.as_ref();
185    let dst = dst.as_ref();
186    eprintln!("$ cp {} {}", src.display(), dst.display());
187
188    let mut _tmp;
189    let mut dst = dst;
190    if dst.is_dir()
191        && let Some(file_name) = src.file_name()
192    {
193        _tmp = dst.join(file_name);
194        dst = &_tmp;
195    }
196    fs::copy(src, dst).map_err(|err| {
197        anyhow!("failed to copy `{}` to `{}`: {err}", src.display(), dst.display())
198    })?;
199
200    Ok(())
201}

203/// Copies the directory `src` to `dst`, which must not exist. Symlinks are not followed.
204pub fn copy_dir(src: &Path, dst: &Path) -> anyhow::Result<()> {
205    eprintln!("$ cp -r {} {}", src.display(), dst.display());
206    copy_dir_inner(src, dst)
207        .map_err(|err| anyhow!("failed to copy `{}` to `{}`: {err}", src.display(), dst.display()))
208}
209
210fn copy_dir_inner(src: &Path, dst: &Path) -> io::Result<()> {
211    fs::create_dir(dst)?;
212    for entry in fs::read_dir(src)? {
213        let entry = entry?;
214        let (src, dst) = (entry.path(), dst.join(entry.file_name()));
215        let file_type = entry.file_type()?;
216        if file_type.is_dir() {
217            copy_dir_inner(&src, &dst)?;
218        } else if file_type.is_symlink() {
219            #[cfg(unix)]
220            std::os::unix::fs::symlink(fs::read_link(&src)?, &dst)?;
221            #[cfg(not(unix))]
222            fs::copy(&src, &dst)?;
223        } else {
224            fs::copy(&src, &dst)?;
225        }
226    }
227    Ok(())
228}

230/// Removes the file or directory at `path`. A symlink is removed itself, not what it points to.
231pub fn remove_path(path: &Path) -> anyhow::Result<()> {
232    match path.symlink_metadata() {
233        Ok(meta) => {
234            if meta.is_dir() { remove_dir_all(path) } else { fs::remove_file(path) }
235                .map_err(|err| anyhow!("failed to remove path `{}`: {err}", path.display()))
236        }
237        Err(err) if err.kind() == io::ErrorKind::NotFound => Ok(()),
238        Err(err) => Err(anyhow!("failed to remove path `{}`: {err}", path.display())),
239    }
240}

242#[cfg(not(windows))]
243fn remove_dir_all(path: &Path) -> io::Result<()> {
244    fs::remove_dir_all(path)
245}

248#[cfg(windows)]
249fn remove_dir_all(path: &Path) -> io::Result<()> {
250    for _ in 0..99 {
251        if fs::remove_dir_all(path).is_ok() {
252            return Ok(());
253        }
254        std::thread::sleep(std::time::Duration::from_millis(10))
255    }
256    fs::remove_dir_all(path)
257}
