1//! Type dependency graph: maps each local struct/enum to the other local
2//! struct/enums referenced in its field definitions.
3
4use std::{io, path::Path};
5
6use rustc_data_structures::unord::UnordMap;
7use rustc_hir::{def::DefKind, def_id::DefId};
8use rustc_middle::ty::{self, TyCtxt};
9
10use crate::dep_graph::dump_dep_graph_json;
11
12/// Collect DefIds of all ADTs (struct/enum) referenced in `ty`, recursively.
13fn collect_adt_refs<'tcx>(ty: ty::Ty<'tcx>, out: &mut Vec<DefId>) {
14    match ty.kind() {
15        ty::TyKind::Adt(adt_def, args) => {
16            out.push(adt_def.did());
17            for arg in args.iter() {
18                if let ty::GenericArgKind::Type(inner_ty) = arg.kind() {
19                    collect_adt_refs(inner_ty, out);
20                }
21            }
22        }
23        ty::TyKind::Array(inner, _) | ty::TyKind::Slice(inner) => {
24            collect_adt_refs(*inner, out);
25        }
26        ty::TyKind::Ref(_, inner, _) | ty::TyKind::RawPtr(inner, _) => {
27            collect_adt_refs(*inner, out);
28        }
29        ty::TyKind::Tuple(tys) => {
30            for inner in tys.iter() {
31                collect_adt_refs(inner, out);
32            }
33        }
34        _ => {}
35    }
36}
37
38pub fn dump_type_graph(tcx: TyCtxt<'_>, dir: &Path) -> io::Result<()> {
39    let local_adts: Vec<DefId> = tcx
40        .iter_local_def_id()
41        .filter(|&local_id| matches!(tcx.def_kind(local_id), DefKind::Struct | DefKind::Enum))
42        .map(|local_id| local_id.to_def_id())
43        .collect();
44
45    let adt_to_idx: UnordMap<DefId, usize> = local_adts
46        .iter()
47        .enumerate()
48        .map(|(i, &did)| (did, i))
49        .collect();
50
51    let adj: Vec<Vec<usize>> = local_adts
52        .iter()
53        .map(|&def_id| {
54            let adt_def = tcx.adt_def(def_id);
55            let mut refs = Vec::new();
56            for variant in adt_def.variants() {
57                for field in &variant.fields {
58                    let field_ty = tcx.type_of(field.did).skip_binder();
59                    collect_adt_refs(field_ty, &mut refs);
60                }
61            }
62            let mut local_refs: Vec<usize> = refs
63                .into_iter()
64                .filter_map(|did| adt_to_idx.get(&did).copied())
65                .collect();
66            local_refs.sort_unstable();
67            local_refs.dedup();
68            let self_idx = adt_to_idx.get(&def_id).copied().unwrap();
69            local_refs.retain(|&idx| idx != self_idx);
70            local_refs
71        })
72        .collect();
73
74    dump_dep_graph_json(tcx, dir, "type_graph.json", &local_adts, &adj)
75}
